% 4. Interruptores de la plantilla
%
%    Una sola fuente LaTeX produce tres variantes del documento. Lo
%    habitual es NO tocar estas líneas y elegir la variante desde el
%    Makefile (make · make annotated · make final), que define la macro
%    correspondiente al compilar. Edítelas sólo si desea fijar una
%    combinación distinta de forma permanente.
%
%      Variante   | instrucciones | ejemplos | observaciones
%      -----------+---------------+----------+---------------
%      trabajo    |      sí       |    sí    |      no
%      anotada    |      sí       |    sí    |      sí
%      final      |      no       |    no    |      no
% ---------------------------------------------------------------------
\newif\ifmostrarinstrucciones   \mostrarinstruccionestrue
\newif\ifmostrarejemplos        \mostrarejemplostrue
\newif\ifmostrarobservaciones   \mostrarobservacionesfalse

\ifdefined\VARIANTEANOTADA      % make annotated
  \mostrarinstruccionestrue
  \mostrarejemplostrue
  \mostrarobservacionestrue
\fi
\ifdefined\VARIANTEFINAL        % make final
  \mostrarinstruccionesfalse
  \mostrarejemplosfalse
  \mostrarobservacionesfalse
\fi

% Etiqueta que identifica la variante en la portada. Queda vacía en la
% versión final, que es la que se entrega y no debe anunciarse como
% plantilla.
\newcommand{\varianteetiqueta}{%
  \ifmostrarobservaciones
    Versión anotada · documenta el origen de las correcciones%
  \else
    \ifmostrarinstrucciones Versión de trabajo · plantilla por rellenar\fi
  \fi}
